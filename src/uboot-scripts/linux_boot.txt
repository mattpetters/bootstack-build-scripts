# Set environment needed for boot
setenv ini_name L4T-focal.ini
setenv ovcfgaddr 0x8d100000
setenv fdtovaddr 0x8d200000
setenv evtmpaddr 0x8d300000

# Load hekate ini and test if necessary variables exists
if load mmc 0:1 bootloader/ini/${ini_name}; then setenv boot_index 0;
elif load mmc 1:1 bootloader/ini/${ini_name}; then setenv boot_index 1;
else reset; fi;

if ini uboot ${fdtovaddr} ${filesize}; then
	if test -n ${device}; echo device type ${device};
	else reset; fi;

	if test -n ${rootfs}; then echo partition number ${rootfs};
	else reset; fi;
	
	# patch dtb with overlays if they exist
	if test -n ${overlays}; then
		echo loaded overlays: ${overlays}
		for ov in ${overlays}; do
		    echo overlaying ${ov}...
			fdt resize 8192

			if load mmc ${boot_index}:1 ${ovcfgaddr} switchroot/overlays/${ov}.dtbo && fdt apply ${ovcfgaddr}; then
				echo Successfully loaded ${ov}.dtbo...

				# Add additional bootargs for UART-B
				if test ${ov} = tegra210-icosa-UART-B-overlay; then setenv bootargs_extra ${bootargs_extra} ${uartb}; echo Enabled UART-B logging; fi
			fi

			# Add additional bootargs for Serial USB
			if test ${ov} = usb_logging; then setenv bootargs_extra ${usblogging} ${bootargs_extra}; echo Enabled usb serial logging; fi

			# Disable rootdev search done by initramfs
			if test ${ov} = auto_rootdev_disable; then setenv bootargs_extra ${bootargs_extra} "auto_rootdev_disable"; fi

			# Override and launch nfs bootargs here
			if test ${ov} = nfs; then
				if load mmc ${boot_index}:1 ${evtmpaddr} switchroot/overlays/nfs.txt && env import -t ${evtmpaddr} ${filesize}; then
					run nfsargs_run
					echo Using NFS root
				fi
			fi
		done
	fi
else
	reset
fi

# Try to load kernel Image from /boot otherwise reset
if ext4load ${device} 0:${rootfs} 0x83000000 /boot/Image; then setenv device_index 0;
elif ext4load ${device} 1:${rootfs} 0x83000000 /boot/Image; then setenv device_index 1;
else reset; fi;

ext4load ${device} ${device_index}:${rootfs} 0x92000000 /boot/initramfs
ext4load ${device} ${device_index}:${rootfs} 0x8d000000 /boot/tegra210-icosa.dtb

fdt addr 0x8d000000 ${filesize}

run kernelcmd_run

usb reset
booti 0x83000000 0x92000000 0x8d000000
