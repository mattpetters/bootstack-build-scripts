setenv ini L4T-bionic.ini
setenv ovcfgaddr 0x8d100000
setenv fdtovaddr 0x8d200000
setenv evtmpaddr 0x8d300000
setenv iniaddr   0x8d400000

if load mmc 1:1 ${iniaddr} bootloader/ini/${ini} && env import -t ${iniaddr} ${filesize}; then
        if test -n ${device}; then echo device type ${device};
        else reset; fi;

        if test -n ${root_mmc_index}; then echo partition number ${root_mmc_index};
        else reset; fi;

        if test -n ${part_num}; then echo partition number ${part_num};
        else reset; fi;

        if test -n ${rootdev}; then echo partition number ${rootdev};
        else reset; fi;

        if test -n ${id}; then echo partition number ${id};
        else reset; fi;
fi

ext4load ${device} ${root_mmc_index}:${part_num} 0x83000000 /boot/Image
ext4load ${device} ${root_mmc_index}:${part_num} 0x92000000 /boot/initramfs
ext4load ${device} ${root_mmc_index}:${part_num} 0x8d000000 /boot/tegra210-icosa.dtb

fdt addr 0x8d000000 ${filesize}

# patch dtb with overlays if they exist
if test -n ${overlays}; then
        echo loaded overlays.txt: ${overlays}
        for ov in ${overlays}; do
            echo overlaying ${ov}...
                fdt resize 8192

                if load mmc ${mmc_index}:1 ${ovcfgaddr} switchroot/overlays/${ov}.dtbo && fdt apply ${ovcfgaddr}; then
                        echo Successfully loaded ${ov}.dtbo...

                        # Add additional bootargs for UART-B
                        if test ${ov} = tegra210-UART-B-overlay; then setenv bootargs_extra ${bootargs_extra} ${uartb}; echo Enabled UART-B logging; fi
                fi

                # Add additional bootargs for Serial USB
                if test ${ov} = usb_logging; then setenv bootargs_extra ${usblogging} ${bootargs_extra}; echo Enabled usb serial logging; fi

                # Disable rootdev search done by initramfs
                if test ${ov} = auto_rootdev_disable; then setenv bootargs_extra ${bootargs_extra} "auto_rootdev_disable"; fi

                # Override and launch nfs bootargs here
                if test ${ov} = nfs; then
                        if load mmc ${mmc_index}:1 ${evtmpaddr} switchroot/overlays/nfs.txt && env import -t ${evtmpaddr} ${filesize}; then
                                run nfsargs_run
                                echo Using NFS root
                        fi
                fi
        done
fi

run kernelcmd_run

usb reset
booti 0x83000000 0x92000000 0x8d000000
