setenv ovcfgaddr 0x8d100000
setenv fdtovaddr 0x8d200000
setenv evtmpaddr 0x8d300000

usb start

# Load uenv from mmc 0 or 1 using prefix builtin variable loaded from coreboot
if load mmc 1:${distro_bootpart} ${fdtovaddr} ${prefix}/uenv.txt; then; setenv devnum 1
elif load mmc 0:${distro_bootpart} ${fdtovaddr} ${prefix}/uenv.txt; then; setenv devnum 0
else echo No boot device found; reset; fi

if env import -t ${fdtovaddr} ${filesize}; then
        if test -n ${hekate_id}; then echo hekate id is ${hekate_id};
        else echo Rootlabel ID missing; reset; fi

	if test -n ${rootdev}; then echo rootdev is ${rootdev};
        else echo Rootdev missing; reset; fi
		
        setexpr pnum gsub ".*[:alpha:](\\d+$)" "\\1" ${rootdev}
fi

if load mmc ${devnum}:${distro_bootpart} 0x83000000 ${prefix}/Image; then
	load mmc ${devnum}:${distro_bootpart} 0x92000000 ${prefix}/initramfs
	load mmc ${devnum}:${distro_bootpart} 0x8d000000 ${prefix}/tegra210-icosa.dtb
elif load ${devtype} ${devnum}:${pnum} 0x83000000 /boot/Image; then
	load ${devtype} ${devnum}:${pnum} 0x92000000 /boot/initramfs
	load ${devtype} ${devnum}:${pnum} 0x8d000000 /boot/tegra210-icosa.dtb
else echo Unable to load kernel Image; reset; fi

# Store dtb filesize
setenv dtb_part_size ${filesize}

# Load dtb in memory
fdt addr 0x8d000000 ${dtb_part_size}

# patch dtb with overlays if they exist
if test -n ${overlays}; then
        echo loaded overlay: ${overlays}
        for ov in ${overlays}; do
            echo overlaying ${ov}...
                fdt resize 8192

                if load mmc ${devnum}:${distro_bootpart} ${ovcfgaddr} switchroot/overlays/${ov}.dtbo && fdt apply ${ovcfgaddr}; then
			echo Successfully loaded ${ov}.dtbo...
                fi

                # Add additional bootargs for UART-B
                if test ${ov} = tegra210-UART-B-overlay; then setenv bootargs_extra ${bootargs_extra} ${uartb}; echo Enabled UART-B logging; fi

                # Add additional bootargs for Serial USB
                if test ${ov} = usb_logging; then setenv bootargs_extra ${usblogging} ${bootargs_extra}; echo Enabled usb serial logging; fi

                # Disable rootdev search done by initramfs
                if test ${ov} = auto_rootdev_disable; then setenv bootargs_extra ${bootargs_extra} "auto_rootdev_disable"; fi

                # Override and launch nfs bootargs here
                if test ${ov} = nfs; then
                        if load mmc ${devnum}:${distro_bootpart} ${evtmpaddr} switchroot/overlays/nfs.txt && env import -t ${evtmpaddr} ${dtb_part_size}; then
                                run nfsargs_run
                                echo Using NFS root
                        fi
                fi
        done
fi

run kernelcmd_run

setenv bootargs ${bootargs} "vpr_resize tegra_fbmem=0x384000@0xf5a00000 "

usb reset
booti 0x83000000 0x92000000 0x8d000000
