# Set defaults env variables. These can be overridden by uenv.txt.
setenv boot_dir ${prefix}
setenv rootdev mmcblk0p2
setenv hekate_id SWR-UBU
setenv rootfstype ext4
setenv rootfs_fw /lib/firmware/
setenv hdmi_fbconsole 1
setenv orientation 3
setenv reboot_action bootloader
setenv rootlabel_retries 1

# Set logging params for each serial type.
setenv uarta "no_console_suspend console=ttyS0,115200,8n1"
setenv uartb "no_console_suspend console=ttyS1,115200,8n1"
setenv uartc "no_console_suspend console=ttyS2,115200,8n1"
setenv usblogging "console=ttyGS0,115200,8n1"

# Set important addresses.
setenv kernaddr 0x83000000
setenv initaddr 0x92000000
setenv fdtraddr 0x8d000000

# Set temp addresses.
setenv enviraddr 0x8d100000
setenv evovtaddr 0x8d200000
setenv fdtovaddr 0x8d300000

# Load enviroment if found.
if load ${devtype} ${devnum}:${distro_bootpart} ${enviraddr} ${boot_dir}/uenv.txt; then env import -t ${enviraddr} ${filesize}; fi

###### If boot_dir was changed from uenv.txt then it will be used to everything from this point on. ######

# Load Kernel, Initramfs and DT. If failed, reset.
if load ${devtype} ${devnum}:${distro_bootpart} ${kernaddr} ${boot_dir}/Image; then echo Kernel loaded; else echo Kernel failed; reset; fi
if load ${devtype} ${devnum}:${distro_bootpart} ${initaddr} ${boot_dir}/initramfs; then echo Initramfs loaded; else echo Initramfs failed; reset; fi
if load ${devtype} ${devnum}:${distro_bootpart} ${fdtraddr} ${boot_dir}/tegra210-icosa.dtb; then echo DT loaded; else echo DT failed; reset; fi

# Set dtb address and size from above.
fdt addr ${fdtraddr} ${filesize}

# Patch dtb and cmdline with overlays if they exist.
if test -n ${overlays}; then
	echo loading overlays: ${overlays}
	for ov in ${overlays}; do
	    echo overlaying ${ov}...

		if load ${devtype} ${devnum}:${distro_bootpart} ${fdtovaddr} ${boot_dir}/overlays/${ov}.dtbo && fdt resize 8192 && fdt apply ${fdtovaddr}; then
			echo Successfully loaded ${ov}.dtbo...

			# Add additional bootargs for UART-A.
			if test ${ov} = tegra210-icosa-UART-A-overlay; then setenv bootargs_extra ${bootargs_extra} ${uarta}; echo Enabled UART-A logging; fi
			# Add additional bootargs for UART-B.
			if test ${ov} = tegra210-icosa-UART-B-overlay; then setenv bootargs_extra ${bootargs_extra} ${uartb}; echo Enabled UART-B logging; fi
			# Add additional bootargs for UART-C.
			if test ${ov} = tegra210-icosa-UART-C-overlay; then setenv bootargs_extra ${bootargs_extra} ${uartc}; echo Enabled UART-C logging; fi
		fi

		# Add additional bootargs for Serial USB.
		if test ${ov} = usb_logging; then setenv bootargs_extra ${usblogging} ${bootargs_extra}; echo Enabled usb serial logging; fi

		# Disable rootdev search done by initramfs.
		if test ${ov} = auto_rootdev_disable; then setenv bootargs_extra ${bootargs_extra} "auto_rootdev_disable"; fi

		# Override and launch nfs bootargs here.
		if test ${ov} = nfs; then
			if load ${devtype} ${devnum}:${distro_bootpart} ${evovtaddr} ${boot_dir}/overlays/nfs.txt && env import -t ${evovtaddr} ${filesize}; then
				run nfsargs_run
				echo Using NFS root
			fi
		fi
	done
fi

# Prepare display panel id from Nyx.
cp.b 0xED838002 0xED838001 0x1
mw.b 0xED838002 0x0 0x1

# Get display panel handle.
if itest.w *0xED838000 == 0xf20; then echo Display is INNO; fdt get value DHANDLE /host1x/dsi/panel-i-720p-6-2 phandle
elif itest.w *0xED838000 == 0xf30; then echo Display is AUO; fdt get value DHANDLE /host1x/dsi/panel-a-720p-6-2 phandle
elif itest.b *0xED838000 == 0x10; then echo Display is JDI; fdt get value DHANDLE /host1x/dsi/panel-j-720p-6-2 phandle
else Display is Unknown; fi

# Set new active display panel handle.
if test -n ${DHANDLE}; then echo Setting Display panel; fdt resize 8192; fdt set /host1x/dsi nvidia,active-panel <$DHANDLE>; fi

# Set kernel cdmline.
setenv bootargs ${bootargs_extra} "root=/dev/${rootdev} rootfstype=${rootfstype} rw \
 firmware_class.path=${rootfs_fw} access=m2 \
 rootlabel_retries=${rootlabel_retries} \
 swr_dir=${boot_dir} \
 pmc_reboot2payload.enabled=1 \
 pmc_reboot2payload.reboot_action=${reboot_action} \
 pmc_reboot2payload.default_payload=reboot_payload.bin \
 pmc_reboot2payload.hekate_config_id=${hekate_id} \
 fbcon=map:${hdmi_fbconsole},rotation:${orientation} consoleblank=0 \
 nvdec_enabled=0 vpr_resize tegra_fbmem=0x400000@0xf5a00000 "

# Reset usb and prepare it for kernel.
usb reset

# Boot kernel.
booti ${kernaddr} ${initaddr} ${fdtraddr}
