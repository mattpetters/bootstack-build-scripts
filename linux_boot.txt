setenv ovcfgaddr 0x8d100000
setenv fdtovaddr 0x8d200000
setenv evtmpaddr 0x8d300000
setenv distro arch

if load mmc 0:1 ${fdtovaddr} switchroot/${distro}/uenv.txt; then setenv mmc_index 0;
elif load mmc 1:1 ${fdtovaddr} switchroot/${distro}/uenv.txt; then setenv mmc_index 1;
else reset; fi;

if env import -t ${fdtovaddr} ${filesize}; then
        if test -n ${rootdev}; then echo rootdev is ${rootdev};
        else echo Rootdev missing; reset; fi

        if test -n ${id}; then echo hekate id is ${id};
        else echo Rootlabel ID missing; reset; fi

	setexpr part_num gsub ".*(\\d)" "\\1" ${rootdev}
fi

usb start

if load mmc 0:${part_num} 0x83000000 /boot/Image; then setenv root_mmc_index 0;
elif load mmc 1:${part_num} 0x83000000 /boot/Image; then setenv root_mmc_index 1;
elif load mmc ${mmc_index}:1 0x83000000 switchroot/${distro}/Image; then
	load mmc ${mmc_index}:1 0x92000000 switchroot/${distro}/initramfs
	load mmc ${mmc_index}:1 0x8d000000 switchroot/${distro}/tegra210-icosa.dtb
else reset; fi;

if test -n ${root_mmc_index}; then
	load mmc ${root_mmc_index}:${part_num} 0x92000000 /boot/initramfs
	load mmc ${root_mmc_index}:${part_num} 0x8d000000 /boot/tegra210-icosa.dtb
else reset; fi;

fdt addr 0x8d000000 ${filesize}

# patch dtb with overlays if they exist
if test -n ${overlays}; then
        echo loaded overlay: ${overlays}
        for ov in ${overlays}; do
            echo overlaying ${ov}...
                fdt resize 8192

                if load mmc ${mmc_index}:1 ${ovcfgaddr} switchroot/overlays/${ov}.dtbo && fdt apply ${ovcfgaddr}; then
			echo Successfully loaded ${ov}.dtbo...
                fi

                # Add additional bootargs for UART-B
                if test ${ov} = tegra210-UART-B-overlay; then setenv bootargs_extra ${bootargs_extra} ${uartb}; echo Enabled UART-B logging; fi

                # Add additional bootargs for Serial USB
                if test ${ov} = usb_logging; then setenv bootargs_extra ${usblogging} ${bootargs_extra}; echo Enabled usb serial logging; fi

                # Disable rootdev search done by initramfs
                if test ${ov} = auto_rootdev_disable; then setenv bootargs_extra ${bootargs_extra} "auto_rootdev_disable"; fi

                # Override and launch nfs bootargs here
                if test ${ov} = nfs; then
                        if load mmc ${mmc_index}:1 ${evtmpaddr} switchroot/overlays/nfs.txt && env import -t ${evtmpaddr} ${filesize}; then
                                run nfsargs_run
                                echo Using NFS root
                        fi
                fi
        done
fi

run kernelcmd_run

usb reset
booti 0x83000000 0x92000000 0x8d000000
